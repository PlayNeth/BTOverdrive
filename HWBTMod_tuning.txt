; HWBTMod_tuning.txt - a lighter car, a livelier camera, more tyre smoke.
; Lines starting with ; are off: remove the ; to try one, add it to switch one off.
;   KEY = 2.5    KEY *= 1.3    KEY#1 *= 1.3 (2nd number after KEY)    @5F9FCF:KEY (only where that code reads it)
; HWBTMod_tuning.log marks every value a line here changed (-> new value).

; --- handling (the exaggerated test, brought back a bit) -------------------
; rotational inertia: lower = snappier turn-in (stock 1500 / 1800 / 1200)
@5FA011:INERTIA *= 0.4
@5FA029:INERTIA#1 *= 0.4
@5FA041:INERTIA#2 *= 0.4
; car body mass: lower = quicker launch, lighter feel (stock 299)
@5F9FCF:MASS *= 0.65

; --- camera ----------------------------------------------------------------
; field of view (stock 65; the speed FOV in HWBTMod.ini adds to it at speed)
@4246F0:FOV *= 1.15
; distance behind the car (stock -11): closer
@5BC6A5:OFFSET#2 *= 0.75
; how much the camera reacts to acceleration and braking (it pulls back / dips
; in): ACTIVE-SCALE is the amount, ACTIVE-LIMIT the most it can move
ACTIVE-SCALE *= 3
ACTIVE-LIMIT *= 2
; sway (sideways swing in turns) and tilt (lean into turns)
SWAY *= 1.6
TILT *= 2
; trailing behind the car (stock 0.7): lower = tighter
;@5BC68C:LAG *= 0.8

; --- tyre smoke and skids ----------------------------------------------------
; per surface: TYRESMOKE's first number is how much smoke (0-255; stock 170 on
; some surfaces, 64, 1 or 0 on others) -> the same strong smoke everywhere
TYRESMOKE = 200
; tyre particles spawned per slip (stock 15)
TIRE_PART_SPAWN_SCALE *= 2
; when the tyres count as skidding: SKID-DATA's first number is the slip where
; skids (marks and smoke) start (stock 0.45 / 0.15 / 0.01). Lower = earlier,
; so smoke shows when attacking turns, not only in big slides
SKID-DATA *= 0.5
